🧾 R2 (RIP Router — Left Side)
enable
conf t

# LAN Interface
interface gigabitEthernet 0/0/0
ip address 192.168.10.1 255.255.255.0
no shutdown

# Serial to R1
interface serial 0/1/0
ip address 10.0.0.1 255.255.255.252
clock rate 64000   # only if DCE
no shutdown

# RIP Configuration
router rip
version 2
no auto-summary
network 192.168.10.0
network 10.0.0.0

end
write memory




🧾 R3 (OSPF Router — Right Side)

enable
conf t

# LAN Interface
interface gigabitEthernet 0/0/0
ip address 192.168.1.1 255.255.255.0
no shutdown

# Serial to R1
interface serial 0/1/0
ip address 10.0.0.6 255.255.255.252
no shutdown

# OSPF Configuration
router ospf 1
network 192.168.1.0 0.0.0.255 area 0
network 10.0.0.4 0.0.0.3 area 0

end
write memory


🧾 R1 (CORE Router — Redistribution Router)


enable
conf t

# Serial to R2
interface serial 0/1/0
ip address 10.0.0.2 255.255.255.252
no shutdown

# Serial to R3
interface serial 0/1/1
ip address 10.0.0.5 255.255.255.252
clock rate 64000   # only if DCE
no shutdown

# RIP Configuration (towards R2)
router rip
version 2
no auto-summary
network 10.0.0.0
redistribute ospf 1 metric 2

# OSPF Configuration (towards R3)
router ospf 1
network 10.0.0.4 0.0.0.3 area 0
redistribute rip subnets

end
write memory
